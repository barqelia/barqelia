import asyncio import logging import sys from aiogram import Bot, Dispatcher, html from aiogram.client.default import DefaultBotProperties from aiogram.enums import ParseMode from aiogram.filters import CommandStart from aiogram.types import Message # Укажите ваш токен от BotFather TOKEN = "ВАШ_ТЕЛЕГРАМ_ТОКЕН" dp = Dispatcher() @dp.message(CommandStart()) async def command_start_handler(message: Message) -> None: """Обработчик команды /start.""" await message.answer(f"Привет, {html.bold(message.from_user.full_name)}!") @dp.message() async def echo_handler(message: Message) -> None: """Обработчик всех остальных текстовых сообщений (эхо-бот).""" try: await message.send_copy(chat_id=message.chat.id) except TypeError: await message.answer("Ошибка при обработке сообщения.") async def main() -> None: # Инициализация бота с парсингом сообщений в HTML bot = Bot(token=TOKEN, default=DefaultBotProperties(parse_mode=ParseMode.HTML)) # Запуск прослушивания обновлений await dp.start_polling(bot) if __name__ == "__main__": logging.basicConfig(level=logging.INFO, stream=sys.stdout) asyncio.run(main())
